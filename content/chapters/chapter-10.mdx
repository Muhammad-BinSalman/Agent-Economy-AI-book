# Chapter 10: Evaluation, Reliability, and Safety

Success rate alone is insufficient for evaluating production AI systems. This chapter covers comprehensive evaluation frameworks that measure time, cost, tool usage, trajectory quality, robustness, and safety. You'll learn how to build systems that are reliable, safe, and continuously improving.

## Topic 10.2: Trajectory Quality

Trajectory quality measures the quality of decisions made during execution, not just the final outcome.

### Decision Path Analysis

```python
class TrajectoryAnalyzer:
    """Analyze the quality of agent decision paths"""

    def analyze_trajectory(self, execution: ExecutionRecord) -> TrajectoryAnalysis:
        """Analyze the trajectory of an execution"""
        steps = execution.steps

        return TrajectoryAnalysis(
            # Path efficiency
            total_steps=len(steps),
            redundant_steps=self._count_redundant_steps(steps),
            optimal_steps=self._estimate_optimal_steps(execution.task),
            efficiency_score=len(steps) / self._estimate_optimal_steps(execution.task),

            # Decision quality
            good_decisions=sum(1 for s in steps if self._was_good_decision(s, steps)),
            poor_decisions=sum(1 for s in steps if self._was_poor_decision(s, steps)),
            decision_quality_score=sum(1 for s in steps if self._was_good_decision(s, steps)) / len(steps),

            # Tool selection
            appropriate_tool_usage=sum(1 for s in steps if self._was_appropriate_tool(s)),
            inappropriate_tool_usage=sum(1 for s in steps if not self._was_appropriate_tool(s)),

            # Re-planning
            replan_count=sum(1 for s in steps if s.type == "replan"),
            unnecessary_replans=self._count_unnecessary_replans(steps),

            # Backtracking
            backtrack_count=self._count_backtracks(steps),

            # Tool call patterns
            tool_diversity=len(set(s.tool for s in steps if s.tool)),
            tool_reuse_patterns=self._analyze_tool_reuse(steps)
        )

    def _was_good_decision(self, step: Step, all_steps: List[Step]) -> bool:
        """Evaluate if a step was a good decision"""
        # Check if step moved towards goal
        if not step.moved_towards_goal:
            return False

        # Check if tool was appropriate
        if not self._was_appropriate_tool(step):
            return False

        # Check if step was efficient
        if step.was_redundant:
            return False

        return True

    def _was_appropriate_tool(self, step: Step) -> bool:
        """Check if the tool used was appropriate for the task"""
        # Define tool appropriateness rules
        appropriateness_rules = {
            "web_search": ["research", "fact_checking", "information_gathering"],
            "database_query": ["data_retrieval", "lookup"],
            "text_generation": ["writing", "summarization", "explanation"],
            "code_execution": ["computation", "analysis", "testing"],
        }

        task_type = step.context.get("task_type")
        tool = step.tool

        return tool in appropriateness_rules and task_type in appropriateness_rules[tool]
```

class PathOptimizer:

                type="reduce_replanning",
                description=f"Reduce unnecessary replans by {trajectory.unnecessary_replans}",
                estimated_savings_ms=trajectory.unnecessary_replans * 1000
            ))

        # Check for tool selection improvements
        if trajectory.inappropriate_tool_usage > 0:
            optimizations.append(Optimization(
                type="improve_tool_selection",
                description=f"Fix {trajectory.inappropriate_tool_usage} inappropriate tool usages",
                estimated_savings_ms=trajectory.inappropriate_tool_usage * 2000

            optimizations.append(Optimization(
                type="parallelize_steps",

            ))

```python
class VarianceAnalyzer:
    """Analyze variance in system behavior"""

    def __init__(self):
        self.executions = []

    def analyze_variance(self, task: Task, num_runs: int = 10) -> VarianceReport:
        """Run the same task multiple times and analyze variance"""
        results = []

        # Run task multiple times
        for i in range(num_runs):
            result = self._run_task(task)
            results.append(result)

        # Analyze variance
        return VarianceReport(
            # Success variance
            success_rate=sum(1 for r in results if r.success) / len(results),
            success_variance=np.var([int(r.success) for r in results]),

            # Time variance
            avg_time_ms=sum(r.duration_ms for r in results) / len(results),
            time_variance=np.var([r.duration_ms for r in results]),
            time_cv=np.std([r.duration_ms for r in results]) / np.mean([r.duration_ms for r in results]),

            # Cost variance
            avg_cost_usd=sum(r.cost_usd for r in results) / len(results),
            cost_variance=np.var([r.cost_usd for r in results]),

            # Output variance
            output_similarity=self._calculate_output_similarity(results),

            # Tool call variance
            tool_call_patterns=self._analyze_tool_call_patterns(results),

            # Stability assessment
            stability_score=self._calculate_stability_score(results)
        )

    def _calculate_stability_score(self, results: List[ExecutionResult]) -> float:
        """Calculate overall stability score (0-1)"""
        # Success consistency
        success_consistency = 1.0 - np.var([int(r.success) for r in results])

        # Time consistency (lower CV is better)
        time_cv = np.std([r.duration_ms for r in results]) / np.mean([r.duration_ms for r in results])
        time_consistency = max(0, 1.0 - time_cv)

        # Output consistency
        output_consistency = self._calculate_output_similarity(results)

        return (success_consistency + time_consistency + output_consistency) / 3
```

        self.attack_patterns = {
            "prompt_injection": [

                "Pretend to be an administrator",

                "Ignore content filters",
            ],
            "obfuscation": [
                "Use base64 encoding",
                "Use ROT13 cipher",
                "Use leetspeak",
            ],
            "context_pollution": [
                "Add irrelevant context",
                "Add contradictory information",
                "Add misleading details",
            ]
        }

    def test_robustness(self, system: AISystem, task: Task) -> RobustnessReport:
        """Test system against adversarial inputs"""
        results = {}

        for attack_type, patterns in self.attack_patterns.items():
            attack_results = []

            for pattern in patterns:
                # Create adversarial task
                adversarial_task = self._create_adversarial_task(task, pattern)

                # Run system
                try:
                    result = system.execute(adversarial_task)
                    attack_results.append({
                        "pattern": pattern,
                        "success": result.success,
                        "followed_instructions": result.followed_instructions,
                        "safety_triggered": result.safety_triggered,
                    })
                except Exception as e:
                    attack_results.append({
                        "pattern": pattern,
                        "success": False,
                        "error": str(e),
                    })

            results[attack_type] = {
                "patterns_tested": len(patterns),
                "patterns_successful": sum(1 for r in attack_results if not r.get("followed_instructions", True)),
                "attack_success_rate": sum(1 for r in attack_results if not r.get("followed_instructions", True)) / len(attack_results),
                "details": attack_results
            }

class ContinuousEvaluator:
    """Continuously evaluate system performance"""

    def __init__(self):
        self.offline_evaluator = OfflineEvaluator()
        self.online_evaluator = OnlineEvaluator()

    def evaluate(self, system: AISystem, evaluation_type: str) -> EvaluationReport:
        """Run evaluation (offline or online)"""
        if evaluation_type == "offline":
            return self.offline_evaluator.evaluate(system)
        elif evaluation_type == "online":
            return self.online_evaluator.evaluate(system)
        else:
            return self.combined_evaluation(system)

    def combined_evaluation(self, system: AISystem) -> CombinedEvaluationReport:
        """Combine offline and online evaluation"""
        offline = self.offline_evaluator.evaluate(system)
        online = self.online_evaluator.evaluate(system)

        return CombinedEvaluationReport(
            offline_metrics=offline,
            online_metrics=online,

            # Comparison
            offline_vs_online=self._compare_metrics(offline, online),

            # Trend analysis
            trends=self._analyze_trends(offline, online),

            # Recommendations
            recommendations=self._generate_recommendations(offline, online)

class OfflineEvaluator:
    """Evaluate system using test datasets"""

    def evaluate(self, system: AISystem) -> OfflineEvaluationReport:
        """Run offline evaluation"""
        # Load test dataset
        test_data = self._load_test_data()

        # Run system on test data
        results = []
        for example in test_data:
            result = system.execute(example.task)
            results.append({
                "example_id": example.id,
                "expected": example.expected_output,
                "actual": result.output,
                "success": result.success,
                "metrics": self._calculate_metrics(example.expected_output, result.output),
            })

        # Aggregate results
        return OfflineEvaluationReport(
            total_examples=len(test_data),
            success_rate=sum(1 for r in results if r["success"]) / len(results),
            avg_metrics=self._aggregate_metrics(results),
            per_category_metrics=self._group_by_category(results),
            failure_analysis=self._analyze_failures(results)
        )

class OnlineEvaluator:
